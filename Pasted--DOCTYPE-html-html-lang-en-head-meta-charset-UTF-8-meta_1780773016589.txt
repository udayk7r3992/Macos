<script>
    // Intro click handler
    document.getElementById('intro').addEventListener('click', function() {
        this.style.opacity = '0';
        setTimeout(() => {
            this.style.display = 'none';
            document.getElementById('desktop').style.display = 'block';
        }, 500);
    });

    // Update time
    function updateTime() {
        const now = new Date();
        const timeString = now.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' });
        document.getElementById('time').textContent = timeString;
    }
    updateTime();
    setInterval(updateTime, 1000);

    // Window management
    const windows = {
        about: document.getElementById('aboutWindow'),
        projects: document.getElementById('projectsWindow'),
        contact: document.getElementById('contactWindow')
    };

    // Open window function
    function openWindow(windowName) {
        const win = windows[windowName];
        if (win) {
            win.style.display = 'block';
            // Bring to front
            win.style.zIndex = Date.now();
        }
    }

    // Close window function
    function closeWindow(windowName) {
        const win = windows[windowName];
        if (win) {
            win.style.display = 'none';
        }
    }

    // Add click handlers to all icons that can open windows
    document.querySelectorAll('[data-window]').forEach(element => {
        element.addEventListener('click', (e) => {
            e.stopPropagation();
            const windowName = element.getAttribute('data-window');
            if (windowName) {
                openWindow(windowName);
            }
        });
    });

    // Add close handlers
    document.querySelectorAll('.window-control.close').forEach(btn => {
        btn.addEventListener('click', (e) => {
            e.stopPropagation();
            const windowId = btn.closest('.window').id;
            const windowName = windowId.replace('Window', '');
            closeWindow(windowName);
        });
    });

    // Draggable windows functionality
    let activeWindow = null;
    let offsetX, offsetY;

    document.querySelectorAll('.window').forEach(window => {
        const header = window.querySelector('.window-header');
        
        header.addEventListener('mousedown', (e) => {
            activeWindow = window;
            offsetX = e.clientX - window.offsetLeft;
            offsetY = e.clientY - window.offsetTop;
            window.style.zIndex = Date.now();
            
            const onMouseMove = (e) => {
                if (activeWindow) {
                    let newX = e.clientX - offsetX;
                    let newY = e.clientY - offsetY;
                    
                    // Keep window in bounds
                    newX = Math.max(0, Math.min(newX, window.innerWidth - activeWindow.offsetWidth));
                    newY = Math.max(40, Math.min(newY, window.innerHeight - 100));
                    
                    activeWindow.style.left = newX + 'px';
                    activeWindow.style.top = newY + 'px';
                }
            };
            
            const onMouseUp = () => {
                activeWindow = null;
                document.removeEventListener('mousemove', onMouseMove);
                document.removeEventListener('mouseup', onMouseUp);
            };
            
            document.addEventListener('mousemove', onMouseMove);
            document.addEventListener('mouseup', onMouseUp);
            e.preventDefault();
        });
    });

    // Refresh button (reload page)
    document.getElementById('refreshDock').addEventListener('click', () => {
        location.reload();
    });

    // Keyboard shortcuts (Cmd+W to close, Cmd+R to refresh)
    document.addEventListener('keydown', (e) => {
        if ((e.metaKey || e.ctrlKey) && e.key === 'w') {
            e.preventDefault();
            // Find active window and close it
            const visibleWindows = document.querySelectorAll('.window[style*="display: block"]');
            if (visibleWindows.length > 0) {
                const lastWindow = visibleWindows[visibleWindows.length - 1];
                const windowName = lastWindow.id.replace('Window', '');
                closeWindow(windowName);
            }
        }
        if ((e.metaKey || e.ctrlKey) && e.key === 'r') {
            e.preventDefault();
            location.reload();
        }
    });

    // Click on desktop to bring nothing to front (just prevent)
    document.getElementById('desktop').addEventListener('click', (e) => {
        if (e.target === document.getElementById('desktop') || e.target.classList.contains('desktop-icons')) {
            // Do nothing, just keep desktop focused
        }
    });
</script>
